# ==========================================
# GET CATEGORY
# ==========================================
async def get_category(
    bot: commands.Bot,
    category_id: int
):
    if not category_id:
        return None, (
            "Category ID is not configured.\n"
            f"ID: `{category_id}`"
        )

    category = bot.get_channel(category_id)

    if category is None:
        try:
            category = await bot.fetch_channel(category_id)
        except discord.NotFound:
            return None, (
                "Category does not exist.\n"
                f"ID: `{category_id}`"
            )
        except discord.Forbidden:
            return None, "I don't have permission to access that category."
        except discord.HTTPException as e:
            return None, f"Failed to access category:\n`{e}`"

    if not isinstance(category, discord.CategoryChannel):
        return None, f"`{category_id}` is not a category."

    return category, None


# ==========================================
# GET TEXT CHANNEL
# ==========================================
async def get_text_channel(
    bot: commands.Bot,
    channel_id: int
):
    if not channel_id:
        return None, "Channel ID is not configured."

    channel = bot.get_channel(channel_id)

    if channel is not None:
        if isinstance(channel, discord.TextChannel):
            return channel, None
        return None, f"`{channel_id}` is not a text channel."

    try:
        channel = await bot.fetch_channel(channel_id)
    except discord.NotFound:
        return None, "The channel no longer exists."
    except discord.Forbidden:
        return None, "I don't have permission to access that channel."
    except discord.HTTPException as e:
        return None, f"Discord API error:\n`{e}`"

    if not isinstance(channel, discord.TextChannel):
        return None, "The channel is not a text channel."

    return channel, None


# ==========================================
# STAFF CHECK
# ==========================================
def is_staff(member: discord.Member) -> bool:
    if member.guild.owner_id == member.id:
        return True

    if member.guild_permissions.administrator:
        return True

    if STAFF_ROLE_ID:
        return any(role.id == STAFF_ROLE_ID for role in member.roles)

    return False


# ==========================================
# MOVE TICKET CHANNEL
# ==========================================
async def move_ticket_channel(
    bot: commands.Bot,
    ticket: discord.TextChannel,
    status: str,
    quantity: str,
    item: str
):
    category_id = CATEGORY_IDS.get(status)

    if not category_id:
        return False, f"No category configured for `{status}`."

    category, error = await get_category(bot, category_id)

    if category is None:
        return False, error

    channel_name = f"{status}．{quantity}．{item}"

    try:
        await ticket.edit(
            name=channel_name,
            category=category,
            reason=f"Order status changed to {status}"
        )
    except discord.Forbidden:
        return False, (
            "I don't have permission to rename or move the ticket channel.\n\n"
            "Please give the bot **Manage Channels**."
        )
    except discord.HTTPException as e:
        return False, f"Discord rejected the channel update:\n`{e}`"

    return True, None


# ==========================================
# UPDATE STATUS TEXT
# ==========================================
def update_status_text(content: str, status: str) -> str:
    if not content:
        return content

    return re.sub(
        r"(status\s*:\s*\*\*__).*?(__\*\*)",
        lambda match: f"{match.group(1)}{status}{match.group(2)}",
        content,
        flags=re.IGNORECASE
    )


# ==========================================
# STATUS BUTTON
# ==========================================
class QueueStatusButton(discord.ui.Button):
    def __init__(self, status: str, label: str):
        super().__init__(
            label=label,
            style=discord.ButtonStyle.secondary,
            custom_id=f"queue_status_{status}"
        )
        self.status = status

    async def callback(self, interaction: discord.Interaction):
        view = self.view

        if not isinstance(view, QueueView):
            await interaction.response.send_message(
                "❌ Queue controls are unavailable.",
                ephemeral=True
            )
            return

        if not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message(
                "❌ Staff only.",
                ephemeral=True
            )
            return

        if not is_staff(interaction.user):
            await interaction.response.send_message(
                "❌ You do not have permission to change order statuses.",
                ephemeral=True
            )
            return

        await view.handle_status(interaction, self.status)

# ==========================================
# QUEUE VIEW
# ==========================================
class QueueView(discord.ui.View):
    def __init__(
        self,
        ticket_channel_id: int,
        quantity: str,
        item: str,
        customer_id: int
    ):
        super().__init__(timeout=None)

        self.ticket_channel_id = ticket_channel_id
        self.quantity = quantity
        self.item = item
        self.customer_id = customer_id

        # Prevent simultaneous status updates
        self.processing_status = False

        self.add_item(
            QueueStatusButton("processing", "processing")
        )
        self.add_item(
            QueueStatusButton("completed", "completed")
        )
        self.add_item(
            QueueStatusButton("cancelled", "cancelled")
        )

    # ==========================================
    # HANDLE STATUS
    # ==========================================
    async def handle_status(
        self,
        interaction: discord.Interaction,
        status: str
    ):
        print(
            f"[QUEUE] Status selected: {status} | "
            f"user={interaction.user} | id={interaction.user.id}"
        )

        # Prevent simultaneous clicks
        if self.processing_status:
            await interaction.response.send_message(
                "⚠️ This order is already being updated.",
                ephemeral=True
            )
            return

        # Prevent changing a finalized order
        if any(button.disabled for button in self.children):
            await interaction.response.send_message(
                "⚠️ This order has already been finalized.",
                ephemeral=True
            )
            return

        self.processing_status = True

        try:
            await interaction.response.defer(ephemeral=True)
        except discord.HTTPException:
            self.processing_status = False
            return

        try:
            # ==================================
            # GET TICKET
            # ==================================
            ticket, error = await get_text_channel(
                interaction.client,
                self.ticket_channel_id
            )

            if ticket is None:
                await interaction.followup.send(
                    f"❌ {error}",
                    ephemeral=True
                )
                return

            # ==================================
            # MOVE TICKET
            # ==================================
            success, error = await move_ticket_channel(
                interaction.client,
                ticket,
                status,
                self.quantity,
                self.item
            )

            if not success:
                await interaction.followup.send(
                    f"❌ Failed to move ticket.\n\n{error}",
                    ephemeral=True
                )
                return

            if status == "cancelled":
                message = (
                    "_ _\n"
                    "_ _            order is cancelled.  "
                    "<a:66cdf480:1531685109055361064>\n"
                    f"_ _          丶 <@{self.customer_id}>\n"
                    "\n"
                    "_ _"
                )
            else:
                message = STATUS_MESSAGES.get(
                    status,
                    "Order status updated."
                )

            await ticket.send(message)

            queue_message = interaction.message

            if queue_message:
                content = update_status_text(
                    queue_message.content,
                    status
                )

                # Disable buttons on final statuses
                if status in ("completed", "cancelled"):
                    for button in self.children:
                        button.disabled = True

                # Keep buttons visible and save their states
                await queue_message.edit(
                    content=content,
                    view=self
                )

            new_status = STATUS_TEXT.get(status, status.upper())

            await interaction.followup.send(
                f"status changed to **{new_status}**.\n"
                f"ticket moved to **{ticket.name}**.",
                ephemeral=True
            )

        except discord.Forbidden as e:
            print(f"[QUEUE ERROR - FORBIDDEN] {e}")

            try:
                await interaction.followup.send(
                    "❌ Discord denied the bot's action.\n"
                    "Please check the bot's permissions.",
                    ephemeral=True
                )
            except discord.HTTPException:
                pass

        except discord.HTTPException as e:
            print(f"[QUEUE ERROR - HTTP] {e}")

            try:
                await interaction.followup.send(
                    f"❌ Discord API error:\n`{e}`",
                    ephemeral=True
                )
            except discord.HTTPException:
                pass

        except Exception as e:
            print(f"[QUEUE ERROR] {type(e).__name__}: {e}")

            try:
                await interaction.followup.send(
                    f"❌ Something went wrong:\n`{type(e).__name__}: {e}`",
                    ephemeral=True
                )
            except discord.HTTPException:
                pass

        finally:
            self.processing_status = False

class Queue(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot

    @app_commands.command(
        name="queue",
        description="Add an order to the queue."
    )
    @app_commands.describe(
        user="Customer",
        item="Item",
        quantity="Quantity",
        amount="Amount",
        mop="Mode of Payment"
    )
    async def queue(
        self,
        interaction: discord.Interaction,
        user: discord.Member,
        amount: int,
        quantity: str,
        item: str,
        mop: str
    ):
        await interaction.response.defer()

        # ==================================
        # ORIGINAL TICKET
        # ==================================
        ticket_channel = interaction.channel

        if not isinstance(ticket_channel, discord.TextChannel):
            await interaction.followup.send(
                "❌ This command must be used inside a text channel.",
                ephemeral=True
            )
            return

        ticket_channel_id = ticket_channel.id

        # ==================================
        # GET QUEUE CHANNEL
        # ==================================
        queue_channel, error = await get_text_channel(
            self.bot,
            QUEUE_CHANNEL_ID
        )

        if queue_channel is None:
            await interaction.followup.send(
                f"❌ Failed to access queue channel.\n\n{error}",
                ephemeral=True
            )
            return

        # ==================================
        # GET NOTED CATEGORY
        # ==================================
        noted_category, error = await get_category(
            self.bot,
            NOTED_CATEGORY_ID
        )

        if noted_category is None:
            await interaction.followup.send(
                f"❌ Failed to access Noted category.\n\n{error}",
                ephemeral=True
            )
            return

        # ==================================
        # MOVE TICKET TO NOTED
        # ==================================
        try:
            await ticket_channel.edit(
                name=f"noted．{quantity}．{item}",
                category=noted_category,
                reason="New order queued"
            )
        except discord.Forbidden:
            await interaction.followup.send(
                "❌ I don't have permission to rename or move the ticket channel.",
                ephemeral=True
            )
            return
        except discord.HTTPException as e:
            await interaction.followup.send(
                f"❌ Failed to move ticket channel:\n`{e}`",
                ephemeral=True
            )
            return

        # ==================================
        # CREATE QUEUE VIEW
        # ==================================
        view = QueueView(
            ticket_channel_id=ticket_channel_id,
            quantity=quantity,
            item=item,
            customer_id=user.id
        )

        # ==================================
        # QUEUE MESSAGE
        # ==================================
        queue_content = (
            "_ _\n"
            f"　　　　 ฅ  　 <#{ticket_channel_id}> \n"
            f"　　　　 [ {quantity} ] 　{item} \n"
            f"　　　　 via  {mop} — **₱ {amount}.00** \n"
            f"　　　　 丶 status : **__noted__** \n"
            f"　　　　<a:001_appletv:1531665532015611935>  :  {user.mention} \n"
            "\n"
            "_ _"
        )

        # ==================================
        # SEND QUEUE MESSAGE
        # ==================================
        try:
            await queue_channel.send(
                content=queue_content,
                view=view
            )
        except discord.Forbidden:
            await interaction.followup.send(
                "❌ I found the queue channel, but I don't have permission to send messages there.",
                ephemeral=True
            )
            return
        except discord.HTTPException as e:
            await interaction.followup.send(
                f"❌ Discord rejected the queue message:\n`{e}`",
                ephemeral=True
            )
            return

        # ==================================
        # CREATE VIEW QUEUE LINK
        # ==================================
        queue_url = (
            f"https://discord.com/channels/"
            f"{interaction.guild.id}/"
            f"{QUEUE_CHANNEL_ID}"
        )

        confirmation_view = discord.ui.View(timeout=None)
        confirmation_view.add_item(
            ViewQueueButton(queue_url)
        )

        # ==================================
        # CONFIRMATION
        # ==================================
        await interaction.followup.send(
            (
                "_ _\n"
                "_ _           kiwi noted ur order.  "
                "<a:f_inouestrawberry:1557076560391962744>\n"
                f"_ _          丶 {user.mention}\n"
                "_ _"
            ),
            view=confirmation_view
        )

    await bot.add_cog(Queue(bot))
